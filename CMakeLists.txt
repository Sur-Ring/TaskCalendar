cmake_minimum_required(VERSION 3.16)
project(TaskCalendar VERSION 1.0 LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
set(TASK_SOURCES Calendar/TaskModel.cpp Calendar/TaskModel.h Calendar/TaskTimeline.cpp Calendar/CalendarWindow.cpp Calendar/CalendarWindow.h Calendar/DesktopController.cpp Calendar/DesktopController.h)
add_executable(TaskCalendar WIN32 main.cpp ${TASK_SOURCES})
target_link_libraries(TaskCalendar PRIVATE Qt${QT_VERSION_MAJOR}::Widgets)
if(WIN32)
 target_link_libraries(TaskCalendar PRIVATE user32)
endif()
if(MSVC)
 target_compile_options(TaskCalendar PRIVATE /utf-8)
endif()
install(TARGETS TaskCalendar RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
 find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
 add_executable(TaskCalendarTests tests/task_calendar_tests.cpp ${TASK_SOURCES})
 target_include_directories(TaskCalendarTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(TaskCalendarTests PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Test)
 if(WIN32)
  target_link_libraries(TaskCalendarTests PRIVATE user32)
 endif()
 if(MSVC)
  target_compile_options(TaskCalendarTests PRIVATE /utf-8)
 endif()
 add_test(NAME TaskCalendarTests COMMAND TaskCalendarTests)
 set_tests_properties(TaskCalendarTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
